# Every CTest case must have a finite timeout so an unexpected deadlock or
# orphaned child cannot occupy a runner indefinitely. Preserve the explicit,
# larger budgets configured above for long integration/SSE/MCP tests.
get_property(ASGEND_REGISTERED_TESTS DIRECTORY PROPERTY TESTS)
foreach(ASGEND_TEST_NAME IN LISTS ASGEND_REGISTERED_TESTS)
  get_test_property("${ASGEND_TEST_NAME}" TIMEOUT ASGEND_TEST_TIMEOUT)
  # get_test_property may report a <variable>-NOTFOUND sentinel.
  if(NOT ASGEND_TEST_TIMEOUT OR "${ASGEND_TEST_TIMEOUT}" MATCHES "NOTFOUND$" OR
     "${ASGEND_TEST_TIMEOUT}" STREQUAL "")
    set_tests_properties("${ASGEND_TEST_NAME}" PROPERTIES TIMEOUT 60)
  endif()
endforeach()
